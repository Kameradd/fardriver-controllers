void FrameParserTask(void* pv) {
    while(1) {
        Read ring buffer byte-by-byte, sync on 0xAA
        Assemble 16-byte frame if enough bytes
        Validate CRC → map ID→addr → copy to telemetry item
        xQueueSend(telemetry_queue, &item, 0)
        Take data_mutex → memcpy to FardriverData → release
    }
}
10. CommandWriteTask (Core 0, prio 7):
void CommandWriteTask(void* pv) {
    while(1) {
        xQueueReceive(write_queue, &cmd, portMAX_DELAY)
        Validate(cmd.addr, cmd.value)
        Construct and send write frame
        Wait up to 100ms for ACK frame
        If critical: readback and compare, retry once on mismatch
        Invoke cmd.callback(success)
    }
}
11. KeepAliveTask (Core 0, prio 5):
void KeepAliveTask(void* pv) {
    TickType_t last_keepalive = 0;
    while(1) {
        vTaskDelayUntil(&last_keepalive, pdMS_TO_TICKS(1000))
        SendRS323Data(0x13, 0x07, 0x5F, 0x5F)
    }
}
12. TelemetryTask (Core 1, prio 6):
void TelemetryTask(void* pv) {
    while(1) {
        if (xQueueReceive(telemetry_queue, &item, pdMS_TO_TICKS(100)) == pdTRUE) {
            Compute speed, SOC, torque, temperatures
            Push to display_queue, log_queue, lora_queue
        }
    }
}
13. SafetyMonitorTask (Core 1, prio 9) — alert-only:
void SafetyMonitorTask(void* pv) {
    TickType_t last = 0;
    while(1) {
        vTaskDelayUntil(&last, pdMS_TO_TICKS(200))  // 5Hz
        Take mutex → check each monitored param against limits
        If violated → increment violation_count
        If violation_count >= 3 → push WARNING alert to display/log/lora
        If back to normal → reset count, push OK alert
    }
}
